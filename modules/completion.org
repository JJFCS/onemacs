* Bounded, project-aware word completion
Completion reads the current buffer and up to twelve other open buffers.
Inside a project it uses that project's buffers; outside one it uses matching
major modes. It never opens a project tree to mine words while you type.

#+begin_src emacs-lisp
(defun onncera-light-buffer-p ()
  "Whether this buffer is suitable for automatic local editing extras."
  (and (not (file-remote-p default-directory))
       (< (buffer-size) onncera-large-buffer-size)))

(defun onncera-completion-buffers ()
  "Return current and bounded, relevant local buffers for Cape."
  (let* ((current (current-buffer))
         (mode major-mode)
         (project (and (not (file-remote-p default-directory)) (project-current nil)))
         (project-bufs (and project (project-buffers project)))
         (candidates
          (cl-remove-if-not
           (lambda (buffer)
             (and (not (eq buffer current))
                  (with-current-buffer buffer
                    (and (onncera-light-buffer-p)
                         (not (string-prefix-p " " (buffer-name)))
                         (derived-mode-p 'prog-mode 'text-mode 'conf-mode)
                         (if project (memq buffer project-bufs) (eq major-mode mode))))))
           (buffer-list))))
    (cons current (seq-take candidates onncera-completion-buffer-limit))))

(defun onncera-complete-all-buffers ()
  "Manually complete words from all ordinary local editing buffers."
  (interactive)
  (require 'cape)
  (let ((cape-dabbrev-buffer-function
         (lambda ()
           (cl-remove-if-not
            (lambda (buffer)
              (with-current-buffer buffer
                (and (onncera-light-buffer-p)
                     (derived-mode-p 'prog-mode 'text-mode 'conf-mode))))
            (buffer-list)))))
    (call-interactively #'cape-dabbrev)))

(defun onncera-tags-capf ()
  "Complete indexed project identifiers after a TAGS table has been selected."
  (when (or (bound-and-true-p tags-file-name) (bound-and-true-p tags-table-list))
    (require 'etags)
    (tags-completion-at-point-function)))

(defun onncera-static-code-capf ()
  "Merge buffer words, language keywords, snippets and configured tags."
  (apply #'cape-wrap-super
         (append (list #'cape-dabbrev #'cape-keyword #'cape-abbrev #'onncera-tags-capf)
                 (when (fboundp 'yasnippet-capf) (list #'yasnippet-capf)))))

(defun onncera-setup-completion ()
  "Retain native CAPFs and add non-LSP fallback sources to this buffer."
  (when (and (locate-library "cape")
             (derived-mode-p 'prog-mode 'text-mode 'conf-mode 'comint-mode 'eshell-mode))
    (require 'cape)
    ;; Prefix/flex matching is cheaper and more natural for identifiers than
    ;; the minibuffer's space-separated Orderless queries.
    (setq-local completion-styles '(basic flex)
                completion-category-overrides nil
                completion-category-defaults nil
                cape-dabbrev-buffer-function #'onncera-completion-buffers
                cape-line-buffer-function #'onncera-completion-buffers
                dabbrev-case-fold-search nil
                dabbrev-case-replace nil
                corfu-auto (and onncera-completion-auto
                                (onncera-light-buffer-p)
                                (not (derived-mode-p 'comint-mode 'eshell-mode))))
    (let ((native (cl-remove-if
                   (lambda (capf)
                     (memq capf '(t cape-file onncera-static-code-capf)))
                   completion-at-point-functions)))
      ;; Native REPL and mode completion retains its own metadata and boundaries.
      ;; Nonexclusive wrappers permit static fallback when a native source has
      ;; no matching candidates. Files stay separate: they use different bounds.
      (setq-local completion-at-point-functions
                  (append (list #'cape-file)
                          ;; Elisp's symbol CAPF is safe to combine with the
                          ;; static sources, so matching built-ins do not hide
                          ;; words or snippets with the same prefix.
                          (when (memq #'elisp-completion-at-point native)
                            (list (cape-capf-super #'elisp-completion-at-point
                                                   #'onncera-static-code-capf)))
                          (mapcar #'cape-capf-nonexclusive
                                  (remq #'elisp-completion-at-point native))
                          (list #'onncera-static-code-capf))))
    (when (fboundp 'corfu-mode) (corfu-mode 1))))

(use-package cape
  :if (locate-library "cape")
  :commands (cape-dabbrev cape-keyword cape-file cape-line cape-dict
             cape-history cape-abbrev cape-elisp-symbol cape-tex cape-sgml cape-emoji)
  :init
  (require 'project)
  (add-hook 'after-change-major-mode-hook #'onncera-setup-completion 90))

(defun onncera-corfu-return ()
  "Accept a selected candidate; otherwise insert the usual newline."
  (interactive)
  (if (>= corfu--index 0)
      (corfu-insert)
    (corfu-quit)
    (newline-and-indent)))

(defun onncera-toggle-auto-completion ()
  "Toggle automatic completion in this buffer, retaining manual completion."
  (interactive)
  (require 'corfu)
  (setq-local corfu-auto (not corfu-auto))
  (corfu-mode -1)
  (corfu-mode 1)
  (message "Automatic completion %s" (if corfu-auto "on" "off")))
