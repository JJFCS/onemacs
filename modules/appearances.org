#+title: Appearance: keep your fonts, spacing and minimal UI

Your compact line height remains controlled by your patched Emacs 31 variable.
No package changes the patch or adds line spacing. Theme packages are optional;
=M-x onncera-switch-theme= can choose any theme you install.

#+begin_src emacs-lisp
;;; appearances.el --- Your existing visual style -*- lexical-binding: t; -*-
(add-to-list 'default-frame-alist '(fullscreen . maximized))
(onncera-apply-fonts)
(add-hook 'after-make-frame-functions #'onncera-apply-fonts)
(advice-add 'enable-theme :after #'onncera-disable-italics)
(when onncera-theme (load-theme onncera-theme t))
(onncera-disable-italics)
(set-face-attribute 'completions-group-title nil :slant 'normal)

;; Final personal overrides, including Customize settings, win over defaults.
(load custom-file t 'nomessage)
(unless (locate-library "corfu")
  (message "Onemacs first run: M-x onncera-sync-packages, then restart Emacs."))
#+end_src
