* Terminals
Your =M-`= ansi-term popup remains available. EAT is under =C-c t e=.

#+begin_src emacs-lisp
(use-package shell-pop
  :if (locate-library "shell-pop")
  :commands shell-pop
  :init
  (setq shell-pop-universal-key "M-`"
        shell-pop-window-position "bottom"
        shell-pop-full-span nil
        shell-pop-term-shell shell-file-name
        shell-pop-window-size 50
        shell-pop-autocd-to-working-dir nil)
  :config
  (setopt shell-pop-shell-type
          '("ansi-term" "*ansi-term*"
            (lambda () (ansi-term shell-pop-term-shell))))
  (with-eval-after-load 'term
    (keymap-set term-raw-map "M-`" #'shell-pop)))

(use-package eat
  :if (locate-library "eat")
  :commands (eat eat-project)
  :init (setq eat-kill-buffer-on-exit t))
#+end_src
