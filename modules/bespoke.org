(defun onncera-smart-beginning-of-line ()
  "Toggle between indentation and the beginning of the line."
  (interactive)
  (let ((old-point (point)))
    (back-to-indentation)
    (when (= old-point (point)) (beginning-of-line))))

(defun onncera-disable-italics (&rest _)
  "Remove italic face slants after a theme has been enabled."
  (interactive)
  (dolist (face (face-list))
    (when (memq (face-attribute face :slant nil t) '(italic oblique))
      (set-face-attribute face nil :slant 'normal))))

(defun onncera-apply-fonts (&optional frame)
  "Apply available preferred fonts to FRAME."
  (with-selected-frame (or frame (selected-frame))
    (when (display-graphic-p)
      (when (find-font (font-spec :family onncera-font))
        (dolist (face '(default fixed-pitch))
          (set-face-attribute face nil :family onncera-font :height onncera-font-height
                              :width 'condensed :weight 'regular)))
      (when (find-font (font-spec :family "CMU Concrete"))
        (set-face-attribute 'variable-pitch nil :family "CMU Concrete"
                            :height onncera-font-height)))))

(defun onncera-switch-theme ()
  "Choose a theme, remove italics and restore your preferred fonts."
  (interactive)
  (let ((theme (intern (completing-read "Load theme: " (custom-available-themes) nil t))))
    (mapc #'disable-theme custom-enabled-themes)
    (load-theme theme t)
    (onncera-disable-italics)
    (onncera-apply-fonts)))

(defun onncera-choose-fonts ()
  "Choose one of your favourite code fonts."
  (interactive)
  (setq onncera-font
        (completing-read "Choose font: "
                         '("JetBrainsMonoNL Nerd Font" "Lexia Mono VF Trial"
                           "LythMonoSquare Nerd Font" "MartianMono Nerd Font Mono")))
  (onncera-apply-fonts))

(defun onncera-conceal-asterisk (candidate)
  "Hide starred buffers until an asterisk is typed into the minibuffer."
  (let ((name (if (consp candidate) (car candidate) candidate))
        (window (active-minibuffer-window)))
    (or (not (string-prefix-p "*" name))
        (and window
             (with-current-buffer (window-buffer window)
               (string-search "*" (minibuffer-contents-no-properties)))))))

(defun onncera-sort-starred-boofers (original prompt &optional default require-match predicate)
  "Filter buffers while retaining ORIGINAL's PROMPT, DEFAULT and PREDICATE.
REQUIRE-MATCH is forwarded unchanged."
  (funcall original prompt default require-match
           (lambda (candidate)
             (and (or (null predicate) (funcall predicate candidate))
                  (onncera-conceal-asterisk candidate)))))
(advice-add 'read-buffer :around #'onncera-sort-starred-boofers)

(defun onncera-buries-boofers ()
  "Choose a buffer to kill, including starred buffers when * is typed."
  (interactive)
  (kill-buffer (read-buffer "Kill buffer: " (current-buffer) t)))

(defun onncera-set-compact-line-height (pixels)
  "Set your patched compact line height to PIXELS."
  (interactive "nCompact line height (pixels): ")
  (unless (boundp 'onncera-line-height-compact-pixels)
    (user-error "This Emacs build does not support your compact line height patch"))
  (unless (and (integerp pixels) (>= pixels 0))
    (user-error "Compact line height must be a nonnegative integer"))
  (setq onncera-line-height-compact-pixels pixels)
  (force-window-update nil)
  (redisplay t)
  (message "Compact line height: %d px" pixels))
(when (boundp 'onncera-line-height-compact-pixels)
  (setq onncera-line-height-compact-pixels 4))

(defun onncera-open-config ()
  "Open your module directory."
  (interactive)
  (dired (expand-file-name "modules/" onncera-config-directory)))

(defun onncera-doctor ()
  "Report this Emacs build, package availability and external tool paths."
  (interactive)
  (with-help-window "*Onemacs doctor*"
    (princ (format "Emacs %s (%s)\nCompact-line patch: %s\nTree-sitter: %s\n\n"
                   emacs-version system-type
                   (if (boundp 'onncera-line-height-compact-pixels) "available" "not in this build")
                   (if (and (fboundp 'treesit-available-p) (treesit-available-p)) "available" "unavailable")))
    (princ "External tools (missing tools are optional):\n")
    (dolist (program '("git" "rg" "ctags" "python3" "R" "node" "go" "rustc"
                       "ruff" "eslint" "black" "prettier" "gdb" "lldb"))
      (princ (format "%-12s %s\n" program (or (executable-find program) "not found"))))
    (princ "\nMissing declared packages:\n")
    (let ((missing (cl-remove-if #'package-installed-p onncera-packages)))
      (princ (if missing (mapconcat #'symbol-name missing ", ") "none")))))
