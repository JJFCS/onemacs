;;; editing.el --- Programming editing tools -*- lexical-binding: t; -*-
(use-package avy
  :if (locate-library "avy")
  :commands (avy-goto-char-timer avy-goto-char avy-goto-line)
  :init (setq avy-timeout-seconds 1.0))
(use-package move-text
  :if (locate-library "move-text")
  :commands (move-text-up move-text-down))
(use-package multiple-cursors
  :if (locate-library "multiple-cursors")
  :commands (mc/mark-previous-like-this mc/mark-next-like-this
             mc/skip-to-previous-like-this mc/skip-to-next-like-this
             mc/mark-all-symbols-like-this-in-defun mc/vertical-align-with-space
             mc/edit-lines)
  :init (setq mc/list-file (onncera-cache-path "mc-lists.el")))
(use-package expand-region
  :if (locate-library "expand-region")
  :commands (er/expand-region er/contract-region))
(use-package easy-kill
  :if (locate-library "easy-kill")
  :commands (easy-kill easy-mark))
(use-package vundo
  :if (locate-library "vundo")
  :commands vundo)
(use-package helpful
  :if (locate-library "helpful")
  :commands (helpful-callable helpful-variable helpful-key helpful-at-point helpful-command))
(use-package imenu-list
  :if (locate-library "imenu-list")
  :commands imenu-list-smart-toggle
  :init (setq imenu-list-position 'right imenu-list-size 0.23 imenu-list-auto-resize nil))
(use-package hl-todo
  :if (locate-library "hl-todo")
  :commands hl-todo-mode)

(defun onncera-programming-extras ()
  "Turn on inexpensive editing aids in normal local source buffers."
  (when (onncera-light-buffer-p)
    (display-line-numbers-mode 1)
    (show-paren-local-mode 1)
    (subword-mode 1)
    (when (fboundp 'hl-todo-mode) (hl-todo-mode 1))
    (when (fboundp 'hs-minor-mode)
      (condition-case nil (hs-minor-mode 1) (error nil)))))

(require 'hideshow)
(setq show-paren-delay 0.1
      imenu-auto-rescan t
      imenu-level-separator "/"
      imenu-max-item-no-type 0
      whitespace-style '(face tabs tab-mark trailing space-before-tab))
(add-hook 'prog-mode-hook #'onncera-programming-extras)

;; Cleanup is explicit. Saving a legacy file should not rewrite unrelated lines.
(defun onncera-clean-whitespace ()
  "Delete trailing whitespace in the current buffer or active region."
  (interactive)
  (if (use-region-p)
      (delete-trailing-whitespace (region-beginning) (region-end))
    (delete-trailing-whitespace)))
