New prefixes use the user-reserved C-c letter space. Mode-local C-c bindings
still take precedence, as in ordinary Emacs. C-x p remains the native project map.

(defun onncera-prefix (key name bindings)
  "Create a named prefix on KEY with NAME and BINDINGS."
  (let ((map (make-sparse-keymap name)))
    (dolist (binding bindings)
      (define-key map (kbd (car binding)) (cdr binding)))
    (global-set-key (kbd key) map)
    map))
